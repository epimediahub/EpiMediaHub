08406     def globalSearchEntered(self, query):
08407         if not query:
08408             return
08409         profile = get_profile(self.profile_id) or {}
08410         if db_source_mode(self.profile_id) == "xtream":
08411             try:
08412                 xtream_load_kind(self.profile_id, "movies")
08413                 xtream_load_kind(self.profile_id, "series")
08414             except Exception as error:
08415                 self.session.open(MessageBox, tr("search_index_error", error=str(error)), MessageBox.TYPE_ERROR, timeout=6)
08416                 return
08417         allowed = {
08418             "movies": set(allowed_category_names(profile, "movies", include_locked=False)),
08419             "series": set(allowed_category_names(profile, "series", include_locked=False)),
08420         }
08421         results = []
08422         for entry in playlist_entries(self.profile_id, query=query):
08423             kind = entry.get("type")
08424             if kind not in ("movies", "series"):
08425                 continue
08426             group = clean_text(entry.get("group")) or "Ohne Kategorie"
08427             if group in allowed.get(kind, set()):
08428                 results.append(entry)
